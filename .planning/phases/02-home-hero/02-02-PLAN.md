### 2. Implement Intro and Scrub Timelines with GSAP Context
- In `useLayoutEffect` / `useEffect` (isomorphic):
  ```tsx
  if (isReducedMotion()) {
    // Instantly reveal completed drawing, wash, stamp, and tracker stage
    return;
  }

  const ctx = gsap.context((self) => {
    // 1. Time-based intro timeline
    const intro = gsap.timeline({ defaults: { ease: 'power2.out' } });
    intro
      .fromTo('.cl', { strokeDashoffset: 1 }, { strokeDashoffset: 0, duration: 1.2, stagger: 0.07 })
      .fromTo('.s1', { strokeDashoffset: 1 }, { strokeDashoffset: 0, duration: 1.1, stagger: 0.14 }, '-=0.6')
      .fromTo(['.hero-copy', '.hero-stages', '#cue'], { opacity: 0, y: 15 }, { opacity: 1, y: 0, duration: 0.8, stagger: 0.1 }, '-=0.4');

    // 2. Scrub timeline with ScrollTrigger pin
    const scrub = gsap.timeline({
      scrollTrigger: {
        trigger: containerRef.current,
        start: 'top top',
        end: '+=300%',
        pin: true,
        scrub: 0.7,
        onUpdate: (self) => {
          const p = self.progress;
          // Stage tracker thresholds: <0.03 Sketch, <0.45 Develop, <0.78 Build, else Handover
          updateStage(p < 0.03 ? 0 : p < 0.45 ? 1 : p < 0.78 ? 2 : 3);
        },
      },
    });

    // Choreograph scrub tweens matching prototype:
    scrub
      .to('#cue', { opacity: 0, duration: 1 }, 0)
      .fromTo('.s2', { strokeDashoffset: 1 }, { strokeDashoffset: 0, stagger: 0.08, duration: 3 }, 0)
      .fromTo('.dim', { strokeDashoffset: 1 }, { strokeDashoffset: 0, stagger: 0.1, duration: 1.8 }, 2.2)
      .fromTo('.note', { opacity: 0 }, { opacity: 1, stagger: 0.15, duration: 1.5 }, 3.0)
      .fromTo('#wash-clip', { attr: { y: 520, height: 0 } }, { attr: { y: 0, height: 520 }, duration: 3.2, ease: 'power1.inOut' }, 4.6)
      .to('.cl', { opacity: 0, duration: 1.5 }, 6.0)
      .fromTo('#stamp', { opacity: 0, scale: 1.4 }, { opacity: 1, scale: 1, ease: 'back.out(2)', duration: 0.8 }, 8.2)
      .fromTo('.hl-path', { strokeDashoffset: 1 }, { strokeDashoffset: 0, duration: 1.2, ease: 'power1.inOut' }, 8.5);
  }, containerRef);

  // Refresh geometry after fonts load
  if (typeof document !== 'undefined' && document.fonts) {
    document.fonts.ready.then(() => ScrollTrigger.refresh());
  }

  return () => ctx.revert();
  ```
